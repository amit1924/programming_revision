Layout Types

File Structure

src/app/
  _layout.tsx    ← root layout
  index.tsx
  (tabs)/
    _layout.tsx ← tab layout
    index.tsx
    feed.tsx
  products/
    _layout.tsx ← stack layout
    index.tsx
    [id].tsx
💡 Rule: Every directory can have a _layout.tsx. The file structure + layout files = complete navigation tree. No manual NavigationContainer needed.

Layout Code Examples

Root Layout — src/app/_layout.tsx
import { Stack } from 'expo-router';
import { useFonts } from 'expo-font';

export default function RootLayout() {
  const [loaded] = useFonts({
    SpaceMono: require('@/assets/fonts/SpaceMono-Regular.ttf'),
  });
  if (!loaded) return null;

  return <Stack />;
}
Stack Layout — app/products/_layout.tsx
import { Stack } from 'expo-router';

export default function StackLayout() {
  return (
    <Stack>
      <Stack.Screen
        name="[id]"
        options={{ headerShown: false }}
      />
    </Stack>
  );
}
Tabs Layout — app/(tabs)/_layout.tsx
import { Tabs } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';

export default function TabLayout() {
  return (
    <Tabs>
      <Tabs.Screen
        name="index"
        options={{
          title: "Home",
          tabBarIcon: ({ color }) =>
            <Ionicons name="home" size={24} color={color} />,
        }}
      />
      <Tabs.Screen name="feed" />
    </Tabs>
  );
}
Slot Layout — app/social/_layout.tsx
import { Slot } from 'expo-router';

export default function Layout() {
  return (

    </>

}
