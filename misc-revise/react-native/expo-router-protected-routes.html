⚠️ Client-side only: Protected routes are evaluated on the client. Not a replacement for server-side auth. During static generation, no HTML is created for protected routes.

Protected Route Patterns

Basic Protected Routes
import { Stack } from 'expo-router';

const isLoggedIn = false;

export default function AppLayout() {
  return (
    <Stack>
      <Stack.Protected guard={!isLoggedIn}>
        <Stack.Screen name="login" />
      </Stack.Protected>
      <Stack.Protected guard={isLoggedIn}>
        <Stack.Screen name="private" />
      </Stack.Protected>
    </Stack>
  );
}
Nesting Protected Screens
import { Stack } from 'expo-router';

const isLoggedIn = true;
const isAdmin = true;

export default function AppLayout() {
  return (
    <Stack>
      <Stack.Protected guard={isLoggedIn}>

        <Stack.Protected guard={isAdmin}>
          <Stack.Screen name="private" />
        </Stack.Protected>

        <Stack.Screen name="about" />

      </Stack.Protected>
    </Stack>
  );
}
Tabs Protected
import { Tabs } from 'expo-router';

const isLoggedIn = false;

export default function TabLayout() {
  return (
    <Tabs>
      <Tabs.Screen name="index" />

      <Tabs.Protected guard={isLoggedIn}>
        <Tabs.Screen name="private" />
        <Tabs.Screen name="profile" />
      </Tabs.Protected>

      <Tabs.Protected guard={!isLoggedIn}>
        <Tabs.Screen name="login" />
      </Tabs.Protected>
    </Tabs>
  );
}
Fallback Screen
import { Stack } from 'expo-router';

const isLoggedIn = false;

export default function AppLayout() {
  return (
    <Stack>
      <Stack.Protected guard={isLoggedIn}>
        <Stack.Screen name="index" />
        <Stack.Screen name="private" />
      </Stack.Protected>
      <Stack.Screen name="login" />
    </Stack>
  );
}
