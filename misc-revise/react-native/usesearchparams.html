Syntax

import { useSearchParams }
  from 'react-router-dom';

const [searchParams, setSearchParams] =
  useSearchParams();

// Read
searchParams.get('key')     // → value | null
searchParams.getAll('key')  // → string[]
searchParams.has('key')    // → boolean

// Write
setSearchParams({ key: 'val' }); // replaces all
setSearchParams(prev => {
  prev.set('key', 'val');
  return prev;
});

How it works

  • URL ?key=value parsed into a URLSearchParams object
  • Calling setSearchParams updates the URL and re-renders
  • Use .get() for single values, .getAll() for duplicate keys
  • Use .has() to check existence
  • Use .toString() to see the full query string
💡 Note: setSearchParams merges with existing params by default (unlike useState). Use the updater function form to modify individual params without losing others.

Minimal Example

ProductsScreen.tsx 
import { useSearchParams } from 'react-router-dom';

function ProductsScreen() {
  const [params, setParams] = useSearchParams();

  // Read
  const search = params.get('search') || '';
  const page = Number(params.get('page')) || 1;
  const category = params.get('category') || '';

  // Write (updater pattern — preserves other params)
  const setSearch = (term) => {
    setParams(prev => {
      if (term) prev.set('search', term);
      else prev.delete('search');
      return prev;
    });
  };

  return (
    <View>
      <Text>Search: {search}</Text>
      <Text>Page: {page}</Text>
      <Text>Category: {category}</Text>
    </View>
  );
}
