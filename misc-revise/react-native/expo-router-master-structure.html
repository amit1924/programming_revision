📂 Basic Stack Stack

Structures

Notation Legend

_layout.tsx — layout file
name.tsx — static route
[param] — dynamic route
(group) — route group
+file.tsx — special route
dir/ — directory
💡 Rule: The src/app/ directory IS your navigation tree. Adding a file = adding a route. _layout.tsx files define how routes relate (Stack, Tabs, etc.).

Key Files per Structure

Stack
// app/_layout.tsx
import { Stack } from 'expo-router';
export default () => <Stack />;
Tabs
// app/(tabs)/_layout.tsx
import { Tabs } from 'expo-router';
export default () => (
  <Tabs>
    <Tabs.Screen name="index" />
    <Tabs.Screen name="feed" />
  </Tabs>
);
Drawer
// app/_layout.tsx
import { Drawer } from 'expo-router/drawer';
export default () => (
  <Drawer>
    <Drawer.Screen name="index" />
  </Drawer>
);
