Navigation Methods

useRouter API

import { useRouter } from 'expo-router';

const router = useRouter();

router.navigate('/about');
router.push('/about');
router.back();
router.replace('/about');
router.setParams({ limit: 50 });
💡 navigate vs push: navigate unwinds to an existing route if it's already on the stack. push always pushes a new page. replace replaces the current page (no back navigation to it).

Navigation Patterns

Link + useRouter
import { Link, useRouter } from 'expo-router';
import { Button } from 'react-native';

export default function Home() {
  const router = useRouter();

  return (
    <>
      <Link href="/about">About</Link>
      <Button
        title="Go"
        onPress={() => router.navigate('/about')}
      />
    </>
  );
}
Dynamic Routes
import { Link, useRouter } from 'expo-router';

// Link with dynamic param inline
<Link href="/user/bacon">View user</Link>

// Link with params object
<Link
  href={{
    pathname: '/user/[id]',
    params: { id: 'bacon' }
  }}
>View user</Link>

// Imperative with params
router.navigate({
  pathname: '/user/[id]',
  params: { id: 'bacon' }
});
Query Params + useLocalSearchParams
// Sending query params
<Link href="/users?limit=20">View 20</Link>

// Receiving page
import { useLocalSearchParams }
  from 'expo-router';

export default function Users() {
  const { id, limit } = useLocalSearchParams();
  // { id: "bacon", limit: "20" }
  return <Text>Users {limit}</Text>;
}

// Update query params without navigation
router.setParams({ limit: 50 });
Redirect + Prefetch
// Redirect (replaces current page)
import { Redirect } from 'expo-router';

export default function Page() {
  return <Redirect href="/about" />;
}

// Prefetch for faster navigation
<Link href="/about" prefetch>
  About
</Link>

// asChild for custom components
<Link href="/other" asChild>
  <Pressable>
    <Text>Home</Text>
  </Pressable>
</Link>
